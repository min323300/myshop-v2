var SS = SpreadsheetApp.getActiveSpreadsheet();

// ============================================================
// 딜러 수수료 구간 상수
// ============================================================
var DEALER_COMMISSION_TIERS = [
  { tier: 1, max: 10000000, hq: 0.6, dealer: 0.4, label: '1구간' },
  { tier: 2, max: 30000000, hq: 0.5, dealer: 0.5, label: '2구간' },
  { tier: 3, max: Infinity, hq: 0.4, dealer: 0.6, label: '3구간' }
];

// ============================================================
// 딜러 구간 계산
// ============================================================
function getDealerTier(sales) {
  for (var i = 0; i < DEALER_COMMISSION_TIERS.length; i++) {
    if (sales < DEALER_COMMISSION_TIERS[i].max) {
      return DEALER_COMMISSION_TIERS[i];
    }
  }
  return DEALER_COMMISSION_TIERS[DEALER_COMMISSION_TIERS.length - 1];
}

// GET 요청
// ============================================================
function doGet(e) {
  var action  = e.parameter.action  || '';
  var storeId = e.parameter.store   || '';

  if (action === 'getPGConfig')               return getPGConfig();
  if (action === 'getSettlement')             return jsonRes(getSettlementData(e.parameter));
  if (action === 'getHqMarginRate')           return jsonRes({ status: 'ok', rate: getHqMarginRate() });
  if (action === 'getConsignmentSettlement')  return jsonRes(getConsignmentSettlement(e.parameter));
  if (action === 'getVatList')                return jsonRes(getVatList(e.parameter));
  if (action === 'getVatSummary')             return jsonRes(getVatSummary(e.parameter));

  // ✅ v4.8 딜러 관련 GET
  if (action === 'getDealerInfo')             return jsonRes(getDealerInfo(e.parameter));
  if (action === 'getDealerCommissionData')   return jsonRes(getDealerCommissionData(e.parameter));
  if (action === 'getDealerPointData')        return jsonRes(getDealerPointData(e.parameter));

  if (storeId)                                return buildOgProxy(storeId);

  return jsonRes({
    status: 'ok',
    message: '담누리마켓 API 정상 v4.8',
    time: Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss')
  });
}

// ✅ v4.8: 딜러 정보 조회
// ============================================================
function getDealerInfo(data) {
  var sheet = SS.getSheetByName('딜러');
  if (!sheet) return { status: 'error', message: '딜러 시트 없음' };

  var rows     = sheet.getDataRange().getValues();
  var headers  = rows[0];
  var dealerId = String(data['딜러ID'] || '').trim();

  for (var i = 1; i < rows.length; i++) {
    var row = {};
    headers.forEach(function(h, idx) { row[h] = rows[i][idx]; });
    if (String(row['딜러ID']).trim() === dealerId) {
      // 현재 구간 계산
      var sales = parseFloat(row['이번달매출'] || 0);
      var tier  = getDealerTier(sales);
      row['현재구간']     = tier.tier;
      row['구간라벨']     = tier.label;
      row['딜러수수료율'] = tier.dealer;
      row['본사수수료율'] = tier.hq;
      return { status: 'ok', dealer: row };
    }
  }
  return { status: 'error', message: '딜러 없음: ' + dealerId };
}

// ============================================================
// ✅ v4.8: 주문 발생 시 딜러 이번달매출 자동 업데이트
//    saveOrder() 내부에서 딜러ID가 dlr_ 로 시작하면 자동 호출
// ============================================================
function updateDealerSales(dealerId, amount) {
  try {
    var sheet = SS.getSheetByName('딜러');
    if (!sheet) return;

    var rows    = sheet.getDataRange().getValues();
    var headers = rows[0];
    var idCol   = headers.indexOf('딜러ID');
    var salesCol= headers.indexOf('이번달매출');
    var tierCol = headers.indexOf('수수료구간');

    if (idCol === -1 || salesCol === -1) return;

    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][idCol]).trim() === dealerId) {
        var curSales = parseFloat(rows[i][salesCol] || 0) + parseFloat(amount || 0);
        var tier     = getDealerTier(curSales);

        sheet.getRange(i + 1, salesCol + 1).setValue(curSales);
        if (tierCol !== -1) {
          sheet.getRange(i + 1, tierCol + 1).setValue(tier.tier);
        }
        return;
      }
    }
  } catch(err) {
    console.log('딜러 매출 업데이트 오류:', err.toString());
  }
}

// ✅ v4.8: 딜러 수수료 계산 및 정산 시트 기록
//    매월 말 또는 수동으로 호출
// ============================================================
function calcDealerCommission(data) {
  var orderSheet = SS.getSheetByName('주문');
  if (!orderSheet) return { status: 'error', message: '주문 시트 없음' };

  var yearMonth = data['yearMonth'] ||
    Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM');
  var dealerId  = data['딜러ID'] || null;

  var rows    = orderSheet.getDataRange().getValues();
  var headers = rows[0];
  var dateCol = headers.indexOf('주문일시');
  var dealCol = headers.indexOf('대리점ID');
  var amtCol  = headers.indexOf('결제금액');
  var statCol = headers.indexOf('주문상태');

  // 딜러별 매출 집계
  var summary = {};
  for (var i = 1; i < rows.length; i++) {
    var rowDate  = String(rows[i][dateCol] || '').substring(0, 7);
    var rowDealer= String(rows[i][dealCol] || '').trim();
    var rowAmt   = parseFloat(rows[i][amtCol]) || 0;
    var rowStat  = String(rows[i][statCol] || '').trim();

    if (rowDate !== yearMonth) continue;
    if (rowStat === '취소' || rowStat === '반품') continue;
    if (!rowDealer.startsWith('dlr_')) continue; // 딜러만
    if (dealerId && rowDealer !== dealerId) continue;

    if (!summary[rowDealer]) summary[rowDealer] = 0;
    summary[rowDealer] += rowAmt;
  }

  var csHeaders = getDealerCommissionHeaders();
  var csSheet   = getOrCreateSheet('딜러정산', csHeaders);
  var results   = [];

  Object.keys(summary).forEach(function(did) {
    var totalSales = summary[did];
    var tier       = getDealerTier(totalSales);
    var dealerComm = Math.round(totalSales * tier.dealer);
    var hqComm     = totalSales - dealerComm;

    // 기존 정산 데이터 업데이트 또는 신규 추가
    var csRows    = csSheet.getDataRange().getValues();
    var csHeaders2= csRows[0];
    var ymCol     = csHeaders2.indexOf('정산월');
    var didCol    = csHeaders2.indexOf('딜러ID');
    var updated   = false;

    for (var j = 1; j < csRows.length; j++) {
      if (String(csRows[j][ymCol]).trim() === yearMonth &&
          String(csRows[j][didCol]).trim() === did) {
        csSheet.getRange(j + 1, 1, 1, csHeaders2.length).setValues([[
          csRows[j][0], yearMonth, did,
          totalSales, tier.tier, tier.label,
          Math.round(tier.dealer * 100) + '%',
          dealerComm, hqComm,
          '정산완료', now()
        ]]);
        csSheet.getRange(j + 1, 1, 1, csHeaders2.length).setBackground('#e8f5e9');
        updated = true;
        break;
      }
    }

    if (!updated) {
      csSheet.appendRow([
        getNextNo(csSheet), yearMonth, did,
        totalSales, tier.tier, tier.label,
        Math.round(tier.dealer * 100) + '%',
        dealerComm, hqComm,
        '정산완료', now()
      ]);
      csSheet.getRange(csSheet.getLastRow(), 1, 1, csHeaders.length)
        .setBackground('#e8f5e9');
    }

    results.push({
      딜러ID:     did,
      정산월:     yearMonth,
      총매출:     totalSales,
      수수료구간: tier.label,
      딜러수수료: dealerComm,
      본사수익:   hqComm
    });
  });

  return { status: 'ok', yearMonth: yearMonth, results: results };
}

// ✅ v4.8: 딜러 수수료 내역 조회
// ============================================================
function getDealerCommissionData(data) {
  var sheet = SS.getSheetByName('딜러정산');
  if (!sheet) return { status: 'ok', rows: [] };

  var rows      = sheet.getDataRange().getValues();
  var headers   = rows[0];
  var dealerId  = String(data['딜러ID']   || '').trim();
  var yearMonth = String(data['yearMonth']|| '').trim();

  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var row = {};
    headers.forEach(function(h, idx) { row[h] = rows[i][idx]; });
    if (dealerId  && String(row['딜러ID']).trim()  !== dealerId)  continue;
    if (yearMonth && String(row['정산월']).trim() !== yearMonth) continue;
    result.push(row);
  }
  return { status: 'ok', rows: result };
}

// ✅ v4.8: 딜러 포인트 적립/차감
// ============================================================
function saveDealerPoint(data) {
  var headers = getDealerPointHeaders();
  var sheet   = getOrCreateSheet('딜러포인트', headers);

  var dealerId = String(data['딜러ID']       || '').trim();
  var type     = String(data['포인트종류']   || '적립').trim(); // 적립 / 사용
  var amount   = parseFloat(data['포인트금액'] || 0);
  var reason   = String(data['사유']          || '').trim();

  if (!dealerId) return { status: 'error', message: '딜러ID 없음' };
  if (amount <= 0) return { status: 'error', message: '포인트 금액 오류' };

  // 사용 시 잔액 확인
  if (type === '사용') {
    var balance = getDealerPointBalance(dealerId);
    if (balance < amount) {
      return { status: 'error', message: '포인트 잔액 부족 (잔액: ' + balance + 'P)' };
    }
  }

  var lastNo = getNextNo(sheet);
  sheet.appendRow([
    lastNo, dealerId, type, amount, reason, data['주문번호'] || '', now()
  ]);

  var color = (type === '사용') ? '#fce4ec' : '#e8f5e9';
  sheet.getRange(sheet.getLastRow(), 1, 1, headers.length).setBackground(color);

  // 딜러 시트 포인트잔액 업데이트
  updateDealerPointBalance(dealerId);

  return { status: 'ok', action: 'created', no: lastNo, type: type, amount: amount };
}

function getDealerPointBalance(dealerId) {
  var sheet = SS.getSheetByName('딜러포인트');
  if (!sheet) return 0;

  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var idCol   = headers.indexOf('딜러ID');
  var typeCol = headers.indexOf('포인트종류');
  var amtCol  = headers.indexOf('포인트금액');

  var balance = 0;
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][idCol]).trim() !== dealerId) continue;
    var amt  = parseFloat(rows[i][amtCol] || 0);
    var type = String(rows[i][typeCol] || '').trim();
    if (type === '사용') balance -= amt;
    else                 balance += amt;
  }
  return balance;
}

function updateDealerPointBalance(dealerId) {
  var sheet = SS.getSheetByName('딜러');
  if (!sheet) return;

  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var idCol   = headers.indexOf('딜러ID');
  var ptCol   = headers.indexOf('포인트잔액');
  if (idCol === -1 || ptCol === -1) return;

  var balance = getDealerPointBalance(dealerId);
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][idCol]).trim() === dealerId) {
      sheet.getRange(i + 1, ptCol + 1).setValue(balance);
      return;
    }
  }
}

// ============================================================
// ✅ v4.8: 매월 1일 딜러 이번달매출 / 수수료구간 자동 리셋
//    → 트리거 등록: setMonthlyResetTrigger() 한번만 실행
// ============================================================
function resetDealerMonthly() {
  var sheet = SS.getSheetByName('딜러');
  if (!sheet) return;

  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var salesCol= headers.indexOf('이번달매출');
  var tierCol = headers.indexOf('수수료구간');

  if (salesCol === -1) return;

  // 전월 정산 먼저 실행
  var lastMonth = getPrevYearMonth();
  calcDealerCommission({ yearMonth: lastMonth });

  // 리셋
  for (var i = 1; i < rows.length; i++) {
    sheet.getRange(i + 1, salesCol + 1).setValue(0);
    if (tierCol !== -1) {
      sheet.getRange(i + 1, tierCol + 1).setValue(1);
    }
  }

  console.log('딜러 월간 리셋 완료: ' + now());
}

function getPrevYearMonth() {
  var now = new Date();
  var y   = now.getFullYear();
  var m   = now.getMonth(); // 0-indexed, 현재 월의 이전달
  if (m === 0) { y -= 1; m = 12; }
  return y + '-' + String(m).padStart(2, '0');
}

// ============================================================
// ✅ v4.8: 월간 리셋 트리거 등록
//    Apps Script 편집기에서 딱 한번만 실행하면 됨
// ============================================================
function setMonthlyResetTrigger() {
  // 기존 트리거 삭제 (중복 방지)
  ScriptApp.getProjectTriggers().forEach(function(t) {
    if (t.getHandlerFunction() === 'resetDealerMonthly') {
      ScriptApp.deleteTrigger(t);
    }
  });

  // 매월 1일 오전 1시에 실행
  ScriptApp.newTrigger('resetDealerMonthly')
    .timeBased()
    .onMonthDay(1)
    .atHour(1)
    .create();

  console.log('월간 리셋 트리거 등록 완료!');
}

// 주문 저장 (v4.8: 딜러 매출 자동 업데이트 연동)
// ============================================================
function saveOrder(data) {
  var headers = getOrderHeaders();
  var sheet   = getOrCreateSheet('주문', headers);

  var firstCell = sheet.getRange(1, 1).getValue();
  if (String(firstCell).trim() !== '주문번호') {
    sheet.insertRowBefore(1);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.getRange(1, 1, 1, headers.length)
      .setFontWeight('bold').setBackground('#1e2336').setFontColor('#ffffff');
    sheet.setFrozenRows(1);
  }

  var orderNo      = String(data['주문번호']     || data['orderNo']      || '').trim();
  var orderDate    = data['주문일시']     || data['orderDate']    || now();

  var phone        = data['연락처']       || data['ordererPhone'] || data['phone']         || '';

  if (!orderNo) return { status: 'error', message: '주문번호 없음' };

  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === orderNo) {
      return { status: 'ok', action: 'already_exists', orderNo: orderNo };
    }
  }

  var newRow = [
    orderNo, orderDate, dealerId,
    ordererName, phone, email,
    receiverName, receiverPhone, address, postCode,
    itemNo, items, qty,
    amount, payMethod,
    referral, memberType, orderStatus, memo,
    now()
  ];

  sheet.getRange(sheet.getLastRow(), 1, 1, newRow.length).setBackground('#fffde7');

  // ✅ v4.8: 딜러 매출 자동 업데이트
  if (String(dealerId).startsWith('dlr_')) {
    updateDealerSales(dealerId, amount);

  var sheet = SS.getSheetByName('주문');

  var rows    = sheet.getDataRange().getValues();

    hCell.setValue('송장번호');
    hCell.setFontWeight('bold').setBackground('#1e2336').setFontColor('#ffffff');
    headers.push('송장번호');
  }

    carrierCol = headers.length;

    hCell2.setFontWeight('bold').setBackground('#1e2336').setFontColor('#ffffff');
    headers.push('택배사');
  }

    pgCol = headers.length;

    hCell3.setFontWeight('bold').setBackground('#1e2336').setFontColor('#ffffff');
    headers.push('PG거래번호');
  }

  var orderNo = String(data['주문번호'] || '').trim();

    '결제대기': '#fffde7', '결제완료': '#e8f5e9',
    '상품준비': '#e3f2fd', '배송중':   '#e8eaf6',

    if (String(rows[i][noCol]).trim() === orderNo) {
      if (data['주문상태'])
        sheet.getRange(i + 1, stCol + 1).setValue(data['주문상태']);

        sheet.getRange(i + 1, 1, 1, sheet.getLastColumn())
          .setBackground(statusColors[data['주문상태']] || '#ffffff');

function saveBulkProducts(data) {
  var products = data.products || [];
  if (!products.length) return { status: 'error', message: '등록할 상품이 없습니다.' };

  var sheet = SS.getSheetByName('상품목록');
  if (!sheet) return { status: 'error', message: '상품목록 시트 없음' };

  // 배송 관련 컬럼 자동 생성
  var deliveryCols = ['배송방법', '배송비', '무료배송조건'];
  var headerRow = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  deliveryCols.forEach(function(col) {
    if (headerRow.indexOf(col) === -1) {
      var newCol = sheet.getLastColumn() + 1;
      sheet.getRange(1, newCol).setValue(col).setFontWeight('bold').setBackground('#e8f4fd');
      headerRow.push(col);
    }
  });

  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  var nextNo = getNextNo(sheet);
  var success = 0, failed = 0, errors = [];

  products.forEach(function(prod, idx) {
    try {
      var rowData = headers.map(function(h) {
        if (h === '번호') return nextNo;
        if (h === '사용여부' && !prod[h]) return 'TRUE';
        if (h === '판매수량' && !prod[h]) return '0';
        if (h === '별점평균' && !prod[h]) return '0';
        if (h === '리뷰수' && !prod[h]) return '0';
        return prod[h] || '';
      });
      sheet.appendRow(rowData);
      nextNo++;
      success++;
    } catch(e) {
      failed++;
      errors.push({ row: idx + 1, message: e.toString() });
    }
  });

  return {
    status: 'ok',
    message: '총 ' + products.length + '건 중 ' + success + '건 등록, ' + failed + '건 실패',
    success: success,
    failed: failed,
    errors: errors
  };
}

function saveRow(sheetName, data, headers) {
  var sheet = getOrCreateSheet(sheetName, headers);
  var no    = data['번호'] ? String(data['번호']).trim() : '';

  if (no) {
    var rows = sheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][0]).trim() === no) {
        var updRow = headers.map(function(h, idx) {
          var newVal = data[h];
          if (newVal === undefined) return rows[i][idx];
          var isDateField = (h === '시작일' || h === '종료일' || h === '시작일시' || h === '종료일시');
          if (isDateField && (newVal === '' || newVal === null)) return rows[i][idx];
          return newVal;
        });
        sheet.getRange(i + 1, 1, 1, headers.length).setValues([updRow]);
        return { status: 'ok', action: 'updated', sheet: sheetName, no: no };
      }
    }
  }

  var lastNo = getNextNo(sheet);
  var newRow = headers.map(function(h) {
    if (h === '번호')   return lastNo;
    if (h === '작성일') return data[h] || new Date().toISOString().split('T')[0];
    return data[h] !== undefined ? data[h] : '';
  });
  sheet.appendRow(newRow);
  return { status: 'ok', action: 'created', sheet: sheetName, no: lastNo };
}

function deleteRow(sheetName, no) {
  var sheet = SS.getSheetByName(sheetName);
  if (!sheet) return { status: 'error', message: sheetName + ' 시트 없음' };

  no = String(no).trim();
  var rows = sheet.getDataRange().getValues();
  for (var i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][0]).trim() === no) {
      sheet.deleteRow(i + 1);
      return { status: 'ok', action: 'deleted', no: no };
    }
  }
  return { status: 'error', message: '해당 번호 없음: ' + no };
}

function saveBulkDealerProducts(data) {
  var products = data.products || [];
  var dealerId = String(data.dealerId || '').trim();
  if (!dealerId) return { status: 'error', message: '대리점ID가 없습니다.' };
  if (!products.length) return { status: 'error', message: '등록할 상품이 없습니다.' };

  var headers = getDealerProductHeaders();
  var sheet   = getOrCreateSheet('대리점상품', headers);

  if (String(sheet.getRange(1, 1).getValue()).trim() !== '번호') {
    sheet.insertRowBefore(1);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#f0f0f0');
  }

  var prefix = getDealerPhonePrefix(dealerId);
  var success = 0, failed = 0, errors = [];

  products.forEach(function(prod, idx) {
    try {
      prod['대리점ID'] = dealerId;
      var nextNo = getNextDealerNo(sheet, prefix, dealerId);
      prod['번호'] = nextNo;
      if (!prod['사용여부']) prod['사용여부'] = 'TRUE';
      if (!prod['위탁여부']) prod['위탁여부'] = 'FALSE';
      if (!prod['본사승인']) prod['본사승인'] = '';
      var newRow = headers.map(function(h) { return prod[h] || ''; });
      sheet.appendRow(newRow);
      success++;
    } catch(e) {
      failed++;
      errors.push({ row: idx + 1, message: e.toString() });
    }
  });

  return {
    status: 'ok',
    message: '총 ' + products.length + '건 중 ' + success + '건 등록, ' + failed + '건 실패',
    success: success,
    failed: failed,
    errors: errors
  };
}

function getDealerPhonePrefix(dealerId) {
  var sheet = SS.getSheetByName('대리점');
  if (!sheet) return '0000';
  var rows     = sheet.getDataRange().getValues();
  var headers  = rows[0];
  var idCol    = headers.indexOf('대리점ID');
  var phoneCol = headers.indexOf('연락처');
  if (idCol === -1 || phoneCol === -1) return '0000';
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][idCol]).trim() === dealerId) {
      var phone = String(rows[i][phoneCol]).replace(/[^0-9]/g, '');
      return phone.slice(-4) || '0000';
    }
  }
  return '0000';
}

function getNextDealerNo(sheet, prefix, dealerId) {
  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var noCol   = headers.indexOf('번호');
  var idCol   = headers.indexOf('대리점ID');
  var maxSeq  = 0;
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][idCol] || '').trim() === dealerId) {
      var rowNo = String(rows[i][noCol] || '').trim();
      if (rowNo.startsWith(prefix)) {
        var seq = parseInt(rowNo.substring(prefix.length)) || 0;
        if (seq > maxSeq) maxSeq = seq;
      }
    }
  }
  return prefix + String(maxSeq + 1);
}

function saveConsignmentSettlement(orderNo, orderRow, orderHeaders) {
  try {
    var dealerIdCol = orderHeaders.indexOf('대리점ID');
    var itemNoCol   = orderHeaders.indexOf('상품번호');
    var dateCol     = orderHeaders.indexOf('주문일시');

    var salesDealerId = String(orderRow[dealerIdCol] || '').trim();
    var itemNos       = String(orderRow[itemNoCol]   || '').trim();
    var orderDate     = String(orderRow[dateCol]     || '').trim();

    var dpSheet = SS.getSheetByName('대리점상품');
    if (!dpSheet) return;

    var dpRows     = dpSheet.getDataRange().getValues();
    var dpHeaders  = dpRows[0];
    var dpNoCol    = dpHeaders.indexOf('번호');
    var dpIdCol    = dpHeaders.indexOf('대리점ID');
    var dpCsCol    = dpHeaders.indexOf('위탁여부');
    var dpAprCol   = dpHeaders.indexOf('본사승인');
    var dpCostCol  = dpHeaders.indexOf('원가');
    var dpPriceCol = dpHeaders.indexOf('가격');
    var dpNameCol  = dpHeaders.indexOf('상품명');

    var nos = itemNos.split(',').map(function(n) { return n.trim(); });

    nos.forEach(function(no) {
      for (var i = 1; i < dpRows.length; i++) {
        if (String(dpRows[i][dpNoCol]).trim() !== no) continue;

        var isConsignment = String(dpRows[i][dpCsCol]  || '').trim().toUpperCase() === 'TRUE';
        var isApproved    = String(dpRows[i][dpAprCol] || '').trim().toUpperCase() === 'TRUE';
        if (!isConsignment || !isApproved) return;

        var consignDealerId = String(dpRows[i][dpIdCol]    || '').trim();
        var cost            = parseFloat(dpRows[i][dpCostCol]  || 0);
        var sellPrice       = parseFloat(dpRows[i][dpPriceCol] || 0);
        var itemName        = String(dpRows[i][dpNameCol]  || '').trim();

        var profit      = sellPrice - cost;
        var hqShare     = Math.round(profit * 0.4);
        var dealerShare = profit - hqShare;

        var vatConsign = Math.round(cost * 0.1);
        var vatHq      = Math.round(hqShare / 11);
        var vatDealer  = Math.round(dealerShare / 11);

        var csSheet = getOrCreateSheet('위탁정산', getConsignmentSettlementHeaders());
        var lastNo  = getNextNo(csSheet);

        csSheet.appendRow([
          lastNo, orderNo, orderDate,
          salesDealerId, consignDealerId, no, itemName,
          sellPrice, cost, profit,
          hqShare, dealerShare,
          vatConsign, vatHq, vatDealer,

        ]);

function calcSettlement(data) {
  var orderSheet = SS.getSheetByName('주문');
  if (!orderSheet) return { status: 'error', message: '주문 시트 없음' };

  var yearMonth = data['yearMonth'] || Utilities.formatDate(new Date(), 'Asia/Seoul', 'yyyy-MM');
  var dealerId  = data['대리점ID'] || null;
  var hqRate    = getHqMarginRate();

  var rows    = orderSheet.getDataRange().getValues();
  var headers = rows[0];
  var dateCol  = headers.indexOf('주문일시');
  var dealCol  = headers.indexOf('대리점ID');
  var amtCol   = headers.indexOf('결제금액');
  var statCol  = headers.indexOf('주문상태');

  var summary = {};
  for (var i = 1; i < rows.length; i++) {
    var rowDate   = String(rows[i][dateCol] || '').substring(0, 7);
    var rowDealer = String(rows[i][dealCol] || '').trim();
    var rowAmt    = parseFloat(rows[i][amtCol]) || 0;
    var rowStat   = String(rows[i][statCol] || '').trim();

    if (rowDate !== yearMonth) continue;
    if (rowStat === '취소' || rowStat === '반품') continue;
    if (rowDealer.startsWith('dlr_')) continue; // 딜러 주문은 별도 정산
    if (dealerId && rowDealer !== dealerId) continue;

    if (!summary[rowDealer]) summary[rowDealer] = 0;
    summary[rowDealer] += rowAmt;
  }

  var settleSheet = getOrCreateSheet('정산', getSettlementHeaders());
  var results = [];

  Object.keys(summary).forEach(function(did) {
    var totalAmt  = summary[did];
    var pgRate    = getDealerPGRate(did);
    var pgFee     = Math.round(totalAmt * pgRate / 100);
    var hqFee     = Math.round(totalAmt * hqRate / 100);
    var dealerAmt = totalAmt - pgFee - hqFee;

    var sRows    = settleSheet.getDataRange().getValues();
    var sHeaders = sRows[0];
    var ymCol    = sHeaders.indexOf('정산월');
    var didCol   = sHeaders.indexOf('대리점ID');
    var updated  = false;

    for (var j = 1; j < sRows.length; j++) {
      if (String(sRows[j][ymCol]).trim() === yearMonth &&
          String(sRows[j][didCol]).trim() === did) {
        var updRow = sHeaders.map(function(h, idx) {
          var map = {
            '정산월': yearMonth, '대리점ID': did,
            '총매출': totalAmt, 'PG요율': pgRate, 'PG수수료': pgFee,
            '본사마진율': hqRate, '본사수수료': hqFee,
            '대리점수령액': dealerAmt, '상태': '정산완료',
            '정산일시': now()
          };
          return map[h] !== undefined ? map[h] : sRows[j][idx];
        });
        settleSheet.getRange(j + 1, 1, 1, sHeaders.length).setValues([updRow]);
        settleSheet.getRange(j + 1, 1, 1, sHeaders.length).setBackground('#e8f5e9');
        updated = true;
        break;
      }
    }

    if (!updated) {
      settleSheet.appendRow([
        getNextNo(settleSheet), yearMonth, did,
        totalAmt, pgRate, pgFee,
        hqRate, hqFee, dealerAmt,
        '정산완료', now()
      ]);
      settleSheet.getRange(settleSheet.getLastRow(), 1, 1, getSettlementHeaders().length)
        .setBackground('#e8f5e9');
    }

    results.push({
      대리점ID: did, 총매출: totalAmt,
      PG수수료: pgFee, 본사수수료: hqFee, 대리점수령액: dealerAmt
    });
  });

  return { status: 'ok', yearMonth: yearMonth, results: results };
}

function getOrCreateSheet(name, headers) {

  if (!sheet) {
    sheet = SS.insertSheet(name);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    sheet.getRange(1, 1, 1, headers.length)
      .setFontWeight('bold').setBackground('#f0f0f0');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function getNextNo(sheet) {
  var last = sheet.getLastRow();
  if (last <= 1) return 1;
  var vals = sheet.getRange(2, 1, last - 1, 1).getValues();
  var max  = 0;
  vals.forEach(function(r) {
    var n = parseInt(r[0]);
    if (!isNaN(n) && n > max) max = n;
  });
  return max + 1;
}

function sendLiveAlarmEmails() {
  var broadcastNo = Browser.inputBox('알림 발송', '방송 번호를 입력하세요 (예: 1)', Browser.Buttons.OK_CANCEL);
  if (broadcastNo === 'cancel') return;

  var liveSheet = SS.getSheetByName('라이브방송');
  if (!liveSheet) { Browser.msgBox('라이브방송 시트 없음'); return; }

  var liveRows     = liveSheet.getDataRange().getValues();
  var liveHeaders  = liveRows[0];
  var liveNoCol    = liveHeaders.indexOf('번호');
  var liveTitleCol = liveHeaders.indexOf('방송제목');
  var liveTimeCol  = liveHeaders.indexOf('시작일시');

  var liveInfo = null;
  for (var i = 1; i < liveRows.length; i++) {
    if (String(liveRows[i][liveNoCol]).trim() === broadcastNo) {
      liveInfo = liveRows[i]; break;
    }
  }
  if (!liveInfo) { Browser.msgBox('방송 번호를 찾을 수 없습니다'); return; }

  var title = liveInfo[liveTitleCol] || '방송';
  var time  = liveInfo[liveTimeCol]  || '';

  var alarmSheet = SS.getSheetByName('라이브알림');
  if (!alarmSheet) { Browser.msgBox('라이브알림 시트 없음'); return; }

  var alarmRows    = alarmSheet.getDataRange().getValues();
  var alarmHeaders = alarmRows[0];
  var noCol        = alarmHeaders.indexOf('방송번호');
  var nameCol      = alarmHeaders.indexOf('이름');
  var emailCol     = alarmHeaders.indexOf('이메일');

  var targets = [];
  for (var j = 1; j < alarmRows.length; j++) {
    if (String(alarmRows[j][noCol]).trim() === broadcastNo) {
      targets.push({ 이름: alarmRows[j][nameCol]||'', 이메일: alarmRows[j][emailCol]||'' });
    }
  }

  if (!targets.length) { Browser.msgBox('알림 신청자가 없습니다'); return; }

  var sent = 0;
  targets.forEach(function(t) {
    if (!t.이메일) return;
    try {
      GmailApp.sendEmail(t.이메일, '[담누리마켓] ' + title + ' 방송 알림',
        t.이름 + '님, 안녕하세요!\n\n"' + title + '" 방송이 곧 시작됩니다.\n\n' +
        '📅 방송 시간: ' + time + '\n📺 바로 보러가기: https://gasway.shop/live.html\n\n담누리마켓 드림');
      sent++;
    } catch(e) { console.log('발송 실패:', t.이메일, e); }
  });

  Browser.msgBox('발송 완료!\n\n총 신청자: ' + targets.length + '명\n이메일 발송: ' + sent + '명');
}
